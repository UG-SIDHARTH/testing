AIM & OBJECTIVE

ALGORITHM / LOGIC HIGHLIGHTS

  8     int items[MAX];
  9 };
 10

 12     s->top = -1;
 13 }
 14

 16     if (s->top >= MAX-1) {
 17         printf("Stack Overflow\n");
 18         return;
 19     }
 20     s->items[++(s->top)] = val;
 21 }
 22

 24     if (s->top == -1) {
 25         printf("Stack Underflow\n");

 27     }
