10struct Stack {

21void push(struct Stack* s, int item) {
22 if (s->top >= MAX - 1) {
23 printf("Stack overflow\n");
24 return;
25 }
26 s->items[++(s->top)] = item;
27}
28 
29int pop(struct Stack* s) {
30 if (s->top == -1) {
31 return -999999;
32 }
33 return s->items[(s->top)--];
34}
